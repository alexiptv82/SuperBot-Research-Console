  def _load_grid_for_block(session_id: str, asset: str) -> pd.DataFrame:
      """Load and concatenate the parquet grid parts for one (session_id, asset).

      Raises:
      - FileNotFoundError: session RAW ZIP not yet imported (case A).
      - BlockDataUnavailableError: ZIP opened but no matching sync_grid_100ms
        parquet entry (case B), empty part list (case C), zero rows after asset
        selection (case D).
      Part concatenation order: filename_ascending.
      """
      zf = open_reference_zip(session_id)   # opens the session ZIP
      with zf:
          grid_names = sorted(
              n for n in zf.namelist()
              if "sync_grid_100ms" in n and n.endswith(".parquet") and asset in n
          )
          # Fallback: try without asset filter
          if not grid_names:
              grid_names = sorted(
                  n for n in zf.namelist()
                  if "sync_grid_100ms" in n and n.endswith(".parquet")
              )
          dfs = []
          for name in grid_names:
              with zf.open(name) as f: buf = io.BytesIO(f.read())
              df = pd.read_parquet(buf)
              if "asset" in df.columns:
                  df = df[df["asset"] == asset].copy()
              dfs.append(df)
      combined = pd.concat(dfs, ignore_index=True)
