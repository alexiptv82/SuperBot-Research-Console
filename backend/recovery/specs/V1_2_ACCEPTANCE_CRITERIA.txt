# =====================================================================
# SECTION C — ORDERED CELL CLASSIFICATION (supersedes former C.6)
# =====================================================================
evaluate_each_applicable_HG_HC_cell_in_this_exact_order=
    1. NOT_APPLICABLE:
         threshold=None on either side (golden or candidate).
    2. CATASTROPHIC_STRUCTURAL:
         (golden_N==0 AND candidate_N>0)
         OR
         (candidate_N==0 AND golden_N>0).
    3. WITHIN_ZERO_ZERO:
         golden_N==0 AND candidate_N==0.
         Classified as WITHIN.
    4. INVALID_RUN:
         required falsified baseline missing.
    5. AXIS_UNVERIFIABLE:
         required control envelope unavailable.
    6. FALSIFIED_BASELINE_ANOMALY:
         F = abs(relN_falsified)
         if F <= env then axis = FAIL.
    7. Otherwise apply band classification:
         WITHIN:        abs(relN) <= env
         CATASTROPHIC:  abs(relN) >= sqrt(env * F)
         ELEVATED:      everything remaining
         (WITHIN takes precedence when both WITHIN and ELEVATED could match;
          CATASTROPHIC takes precedence over ELEVATED but not over WITHIN.)
epsilon_policy=NO_EPSILON
hidden_tolerance_policy=FORBIDDEN

# =====================================================================
# SECTION D — ENVELOPE RATIO r AND MEDIAN
# =====================================================================
envelope_ratio_definition=
    if env >  0                     : r = abs(relN) / env
    if env == 0 AND abs(relN) == 0  : r = 0
    if env == 0 AND abs(relN) >  0  : r = +infinity
    if structural CATASTROPHIC cell : r = +infinity
median_definition=
    odd  N: central sorted value
    even N: arithmetic mean of the two central values
NOT_APPLICABLE_cell_treatment=
    excluded from median computation;
    remains in inventory.

# SECTION E — HG VERDICT ARCHITECTURE

HG_verdict=
    PASS:
        component_1 verdict = PASS
        AND component_2 verdict = PASS
    FAIL:
        component_1 FAILS
        OR component_2 FAILS
        OR any cell classified FALSIFIED_BASELINE_ANOMALY
    PARTIAL:
        otherwise, provided:
            no CATASTROPHIC cell AND
            median(r) over applicable cells <= 1
HG_component_1_PASS_requires=
    no CATASTROPHIC cell
    AND no FALSIFIED_BASELINE_ANOMALY
    AND median(r) <= 1
    AND no INVALID_RUN

        at least one discriminating control cell exists

        OR I12 violation anywhere

        discriminating cells exist
